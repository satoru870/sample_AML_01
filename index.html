// 不審項目のタップ判定ロジック
function checkCDDItem(el, isSuspicious, note) {
if (isSuspicious) {
if (!el.classList.contains('flagged-success')) {
el.classList.add('flagged-success');
alert(`⭕️ 正解！不審な点を検出しました：\n${note}`);
skillScores.cdd += 10;
}
} else {
el.classList.add('flagged-fail');
alert("❌ この項目は一般的な記載内容であり、特に不審点はありません。");
setTimeout(() => el.classList.remove('flagged-fail'), 1200);
}
}
function getRiskBadgeClass(risk) { return risk === 'HIGH' ? 'badge--high' : (risk === 'MEDIUM' ? 'badge--medium' : 'badge--low'); }
function execIDCheck(idx) {
document.getElementById(`id-res-${idx}`).classList.add('active');
const btn = document.getElementById(`btn-id-${idx}`);
btn.innerText = "✓ 本人確認完了"; btn.disabled = true;
skillScores.kyc += 20;
}
function upgradeRisk(idx) {
const badge = document.getElementById(`risk-badge-${idx}`);
badge.className = "badge badge--high";
badge.innerText = "HIGH (手動変更)";
alert("🚨 窓口担当者の判断により、顧客リスク評価を【HIGH】へ手動引き上げました。これによりEDD（詳細調査）の実行が可能になります。");
skillScores.risk += 15;
}
function startTimer() {
clearInterval(timerInterval);
timeLeft = 90;
updateTimerDisplay();
timerInterval = setInterval(() => {
timeLeft--;
updateTimerDisplay();
if (timeLeft <= 0) {
clearInterval(timerInterval);
alert("⏱️ 制限時間切れです！窓口の応対時間を考慮し、迅速に判断してください。");
handleJudge(true, false);
}
}, 1000);
}
function updateTimerDisplay() {
const mins = Math.floor(timeLeft / 60);
const secs = timeLeft % 60;
document.getElementById('timer-text').innerText = `${String(mins).padStart(2, '0')}:${String(secs).padStart(2, '0')}`;
}
function execScr(idx) {
document.getElementById(`scr-res-${idx}`).classList.add('active');
const btn = document.getElementById(`btn-scr-${idx}`);
btn.innerText = "✓ リスト照合完了"; btn.disabled = true;
const boStatusEl = document.getElementById(`bo-status-${idx}`);
if (boStatusEl) {
const currentCase = caseData[idx];
if (currentCase.id === 5) {
boStatusEl.innerHTML = `照合状況: <span style="color:var(--color-red); font-weight:800;">🚨 【反社会的勢力DB HIT】 暴力団関係者リスト一致！</span>`;
} else {
boStatusEl.innerHTML = `照合状況: <span style="color:var(--color-green); font-weight:800;">クリア（反社・PEPヒットなし）</span>`;
}
}
skillScores.kyc += 10;
}
function execEDD(idx) {
document.getElementById(`edd-res-${idx}`).classList.add('active');
const btn = document.getElementById(`btn-edd-${idx}`);
btn.innerText = "✓ EDD調査完了"; btn.disabled = true;
const currentCase = caseData[idx];
document.getElementById(`avatar-display-${idx}`).innerText = currentCase.avatarAngry;
if (!currentCase.isHighRisk) {
alert("⚠️【過剰調査注意】この取引は低リスクな正常取引です。不要なリスク引き上げやEDD（追加調査）を行うと顧客体験悪化・クレームを招きます。");
skillScores.edd -= 10;
} else {
skillScores.edd += 20;
}
}
function openReasonModal() {
const currentCase = caseData[currentCaseIdx];
const container = document.getElementById('reason-buttons-container');
container.innerHTML = '';
currentCase.reasons.forEach(reason => {
const btn = document.createElement('button');
btn.className = 'btn-reason';
btn.innerText = reason.text;
btn.onclick = () => {
document.getElementById('modal-reason').classList.remove('active');
handleJudge(true, reason.correct);
};
container.appendChild(btn);
});
document.getElementById('modal-reason').classList.add('active');
}
function handleJudge(userChoice, isReasonCorrect = true) {
clearInterval(timerInterval);
const currentCase = caseData[currentCaseIdx];
let isCorrect = (userChoice === currentCase.correctChoice);
if (userChoice === true && !isReasonCorrect) isCorrect = false;
if (isCorrect) {
score++; skillScores.cdd += 15; skillScores.risk += 20;
} else {
life--;
}
updateHeader();
const modalTitle = document.getElementById('modal-explain-title');
const modalBody = document.getElementById('modal-explain-body');
const modalAftermath = document.getElementById('modal-explain-aftermath');
if (isCorrect) {
modalTitle.innerText = "⭕️ 正解！"; modalTitle.className = "modal__header modal__header--correct";
} else {
modalTitle.innerText = "❌ 不正解..."; modalTitle.className = "modal__header modal__header--wrong";
}
modalBody.innerHTML = currentCase.explain;
modalAftermath.innerHTML = currentCase.aftermath;
document.getElementById('modal-explain').classList.add('active');
}
function handleModalNext() {
document.getElementById('modal-explain').classList.remove('active');
if (life <= 0 || currentCaseIdx >= caseData.length - 1) {
showFinalResult();
} else {
document.getElementById(`case-panel-${currentCaseIdx}`).classList.remove('active');
currentCaseIdx++;
document.getElementById(`case-panel-${currentCaseIdx}`).classList.add('active');
updateHeader();
startTimer();
}
}
function updateHeader() {
document.getElementById('progress-text').innerText = `${currentCaseIdx + 1}/${caseData.length}`;
document.getElementById('score-text').innerText = score;
document.getElementById('life-text').innerText = "❤️".repeat(Math.max(0, life)) + "🖤".repeat(Math.max(0, 3 - life));
}
